$ cat main.lua
-- list: print the entries of a directory, using LuaFileSystem (a C module).
local lfs = require("lfs")

local dir = arg[1] or "."
local names = {}
for name in lfs.dir(dir) do
    if name ~= "." and name ~= ".." then
        names[#names + 1] = name
    end
end
table.sort(names)
print(table.concat(names, "  "))
[exit 0]
