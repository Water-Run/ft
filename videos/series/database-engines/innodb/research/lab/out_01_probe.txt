CREATE DATABASE innodb_demo DEFAULT CHARACTER SET latin1;
USE innodb_demo;
CREATE TABLE users (
  id INT NOT NULL,
  name CHAR(10) NOT NULL,
  age TINYINT NOT NULL,
  PRIMARY KEY (id),
  KEY idx_name (name)
) ENGINE=InnoDB ROW_FORMAT=DYNAMIC;
