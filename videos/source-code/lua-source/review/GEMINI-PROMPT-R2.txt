附录 A：开场与第 02 章用的 28 行程序，以及它在 Lua 5.5.1 上的输出

```lua
local function make_counter(start)
  local state = {value = start}
  return function(delta)
    state.value = state.value + delta
    return state.value
  end, state
end

local next_value, state = make_counter(40)
setmetatable(state, {
  __index = function(_, key)
    if key == "label" then return "counter" end
  end
})
print(state.label, next_value(2), next_value(3))

local co = coroutine.create(function()
  coroutine.yield(state.value)
  return next_value(1)
end)
local ok1, first = coroutine.resume(co)
print("yield", ok1, first, coroutine.status(co))
local ok2, last = coroutine.resume(co)
print("return", ok2, last, coroutine.status(co))
local ok3, err = pcall(function() error("counter-error", 0) end)
print("protected", ok3, err)
assert(first == 45 and last == 46 and ok1 and ok2 and not ok3)
```
输出：
```

yield	true	45	suspended
return	true	46	dead
protected	false	counter-error
```

--- 08 调用 ---
```lua
-- 08 调用：Lua 调 Lua 不占 C 栈；经过 C 函数就占；尾调用不占新帧
local function deep(n)
  if n == 0 then return 0 end
  return 1 + deep(n - 1)
end
print(deep(100000))

local function probe(lo, hi)            -- 找出非尾递归能走多深
  while lo < hi do
    local mid = (lo + hi + 1) // 2
    if pcall(deep, mid) then lo = mid else hi = mid - 1 end
  end
  return lo
end
local max = probe(100000, 1000000)
print("deepest", max)
print(select(2, pcall(deep, max + 1)))

local level = 0
local function viapcall()
  level = level + 1
  local ok, err = pcall(viapcall)
  if not ok then error(err, 0) end
end
print(pcall(viapcall))
print("level", level)

local function loop(n)
  if n == 0 then return "done" end
  return loop(n - 1)
end
print(loop(10000000))
```
