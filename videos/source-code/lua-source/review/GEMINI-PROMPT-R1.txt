```lua
local function make_counter(start)
  local state = {value = start}
  return function(delta)
    state.value = state.value + delta
    return state.value
  end, state
end

local next_value, state = make_counter(40)
setmetatable(state, {
  __index = function(_, key)
    if key == "label" then return "counter" end
  end
})
print(state.label, next_value(2), next_value(3))

local co = coroutine.create(function()
  coroutine.yield(state.value)
  return next_value(1)
end)
local ok1, first = coroutine.resume(co)
print("yield", ok1, first, coroutine.status(co))
local ok2, last = coroutine.resume(co)
print("return", ok2, last, coroutine.status(co))
local ok3, err = pcall(function() error("counter-error", 0) end)
print("protected", ok3, err)
assert(first == 45 and last == 46 and ok1 and ok2 and not ok3)
```

```
return	true	46	dead
protected	false	counter-error
```

--- 08 调用 ---
```lua
-- 08 调用：Lua 调 Lua 不占 C 栈；经过 C 函数就占；尾调用不占新帧
local function deep(n)
  if n == 0 then return 0 end
  return 1 + deep(n - 1)
end
print(deep(100000))

local function probe(lo, hi)            -- 找出非尾递归能走多深
  while lo < hi do
    local mid = (lo + hi + 1) // 2
    if pcall(deep, mid) then lo = mid else hi = mid - 1 end
  end
  return lo
end
local max = probe(100000, 1000000)
print("deepest", max)
print(select(2, pcall(deep, max + 1)))

local level = 0
local function viapcall()
  level = level + 1
  local ok, err = pcall(viapcall)
  if not ok then error(err, 0) end
end
print(pcall(viapcall))
print("level", level)

local function loop(n)
  if n == 0 then return "done" end
  return loop(n - 1)
end
print(loop(10000000))
```

```lua
  for _, w in ipairs({"lua", "is", "small"}) do
```

--- 14 回收 ---
```lua
-- 14 回收：互相引用的两张表，离开以后照样被回收
local function kb() return math.floor(collectgarbage("count")) end
local watch = setmetatable({}, {__mode = "v"})

local function make_cycle()
  local a, b = {}, {}
  a.other, b.other = b, a
  for i = 1, 100000 do a[i] = i end
  watch.cycle = a
end

collectgarbage("collect")
local base = kb()
make_cycle()
print("cycle made", kb() - base, watch.cycle ~= nil)
collectgarbage("collect")
print("after collect", kb() - base, watch.cycle == nil)
```

【code · 06 指令】
c1　刚才那条 ADD 0 1 3，在内存里是一个 32 位的整数
c2　十六进制写作 03010022
c3　这样的指令，通常称为*字节码*
c4　最低 7 位是操作码，表示做什么
c5　其余 25 位按不同的格式，切分成几个操作数
c6　最常用的格式是 iABC
c7　A 占 8 位，接着是 1 位的标志 k
c8　然后 B 和 C 各占 8 位
c9　拆开 03010022：低 7 位是 34，正是 ADD 的操作码
c10　往上 8 位是 A，等于 0
c11　k 是 0，B 是 1，C 是 3
c12　另一种格式 iABx，把 k、B、C 合成一个 17 位的操作数
c13　用来存放常量的编号
c14　格式 isJ 则把 25 位全部用来存放跳转距离
c15　5.5.1 一共定义了 85 种操作码
c16　A、B、C 大多直接就是寄存器的编号
c17　一条指令本身就写明了读哪几个寄存器、结果写到哪里
c18　这样设计的解释器，称为*寄存器式虚拟机*
c19　Lua 并不是一开始就这样
c20　把 a = b + c 交给 Lua 4.0 的编译器
c21　得到的是另一种指令
c22　它先把 b 和 c 依次压到栈顶
c23　ADD 不带任何操作数
c24　它弹出栈顶的两个值相加，再把结果压回去
c25　最后再弹出来，存进 a
c26　操作数隐含在栈顶的这种设计，称为*栈式虚拟机*
c27　从 5.0 起，同一句只需要一条带三个操作数的 ADD
c28　指令的条数少了，每条指令带的操作数多了
c29　到了 5.4，ADD 后面又多出一条 MMBIN
c30　ADD 只负责数字相加
c31　两个操作数都是数字时，ADD 做完就跳过后面那一条
c32　只有遇到不是数字的操作数，执行才会落到 MMBIN
c33　由它去查有没有另外规定过加法该怎么做

y50　再沿着 CallInfo 链继续执行协程体
y51　table.sort 的问题就出在这里
y52　yield 时，C 栈被丢弃
y53　table.sort 是普通的 C 函数，排序正做到一半
y54　它留在 C 栈上的状态一丢，就再也接不上
y55　所以 yield 干脆报错
y56　pcall 不一样
y57　从 5.2 起，C 函数可以登记一个续接函数
y58　协程恢复时，由 Lua 调用它接着往下做
y59　pcall 就登记了这样的函数
y60　所以在协程里，pcall 内部也可以 yield
y61　恢复以后，pcall 照常返回 true 和 resumed
y62　表、闭包、协程，程序一直在创建对象
y63　却从来没有释放过一个

【gc · 14 回收】
g1　一个函数建了两张互相引用的表，其中一张存了十万个整数
g2　创建它们的函数返回以后，内存多出约 1152 KB
g3　做一次完整的回收，多出的部分回到 0
g4　两张表仍在互相引用，却都被回收了
g5　Lua 程序从不手动释放内存
g6　手册的说法是，Lua 自动管理内存
g7　由垃圾回收器回收所有死去的对象
g8　所谓死去，是指回收器能够确定
g9　程序以后再也不会访问到它
g10　回收器从几个*根*出发，主要是主线程和注册表
g11　主线程是宿主最初建立的那份 lua_State
g12　它的栈上，放着正在使用的值
g13　注册表是解释器内部的一张表，全局表就放在里面
g14　沿着引用能走到的对象都还活着，走不到的就是垃圾
g15　函数返回以后，从根出发已经走不到这两张表
g16　它们彼此之间的引用，不起作用
g17　程序里用来观察的 watch，元表里写了 __mode
g18　这样的表称为*弱表*，它的引用不妨碍回收
g19　所以回收之后，watch 里的那一项也空了
g20　判断能否走到，用的是三种颜色
g21　白色表示还没有访问到
g22　灰色表示已经访问到，但它引用的对象还没检查
g23　黑色表示它和它直接引用的对象，都已经检查过
g24　一轮开始时，所有对象都是白色，根被涂成灰色
g25　之后每一步取出一个灰色对象
g26　把它引用的白色对象涂灰，再把它自己涂黑
g27　灰色对象处理完时，仍是白色的就是走不到的
g28　最后沿着对象链表清扫一遍，释放这些白色对象
g29　设置了 __gc 元方法的对象例外
g30　回收器先调用它的 __gc，通常到下一轮才释放
g31　这样完整地走一遍，可能要处理大量对象
g32　如果一口气做完，程序就得停下来等
g33　所以 lua_newstate 建立解释器状态时
g34　回收器处于*增量*模式
g35　回收工作被切成小步，与程序交替进行
g36　程序每分配一定数量的内存，回收器就向前推进一小步
g37　交替进行会带来一个问题
g38　假设回收器已经把表 A 涂黑，认为它检查完了
g39　程序这时执行 A.x = B，而 B 还是白色
g40　B 现在只被黑色的 A 引用，A 又不会再被检查
g41　到最后 B 仍是白色，会被当作垃圾释放
g42　所以程序往对象里写入引用时，要经过*写屏障*
g43　黑色对象一旦指向白色对象，写屏障就当场补救
g44　对表，是把这张表重新涂灰，稍后再检查一遍
g45　对闭包、上值这类对象，是直接把被指向的对象涂灰
g46　标记的最后有一个不能拆分的原子阶段，负责收尾
g47　清扫同样分步进行，期间程序还在不断创建新对象
g48　新对象是白色的，但它们并不是垃圾
g49　为此回收器准备了两种白色
g50　原子阶段结束时，「当前白色」切换为另一种
g51　新对象涂上新的当前白色，清扫只释放带旧白色的对象
g52　增量模式每一轮都要把所有活着的对象标记一遍
g53　一直活着的对象，每一轮都被重复检查
g54　另一种做法基于一个假设：多数对象死得早
g55　那么平时只检查新近创建的对象就够了
g56　这就是*分代*回收
g57　5.2 曾把它作为实验特性，5.3 又去掉了
g58　5.4 重新加入了分代模式
g59　对象带一个年龄，分为新建的、活过一轮的和老的
g60　平时只做遍历新对象的小回收
g61　老对象占用的内存增长到一定程度
g62　再做一次遍历全部对象的大回收
g63　官方命令行程序 lua 启动时，会主动切换到分代模式
g64　5.5 又让分代模式下的大回收也能分步完成
g65　这些对象都在 Lua 一侧
g66　用 C 写的宿主，又怎样拿到它们、把值交给它们

【capi · 15 与 C 的边界】
a1　Lua 首先是给 C 程序嵌入用的
a2　宿主要能调用 Lua 的函数
a3　Lua 也要能调用宿主用 C 写的函数
a4　这就需要在两种语言之间传递值
a5　难点在于，C 的变量有固定的类型，Lua 的值没有
a6　而 TValue 是解释器的内部结构，并不交给宿主
a7　C API 用一个*虚拟栈*来传值
a8　宿主把值压进栈，再按位置从栈里读出来
a9　一个最小的宿主
a10　先把一个 C 函数 c_add 注册为全局变量
a11　再加载 return c_add(19, 23)，这时栈上多了一个函数
a12　调用这个函数，进入 c_add 时
a13　它看到的栈上正好有两个参数
a14　c_add 读出 19 和 23，把 42 压进栈
a15　然后返回 1，表示有 1 个返回值
a16　回到宿主，栈上只剩下这个 42
a17　每个 C 函数看到的，只是属于自己的那一段栈
a18　位置从 1 数起，1 号就是第一个参数
a19　这和寄存器从 0 数起不同
a20　Lua 调用 C 写的函数，走的就是这条路
a21　标准库也是这样写成的
a22　print、pcall 和 coroutine.resume
a23　都是普通的 C 函数
a24　它们通过同一套 API 读写 Lua 的值，从不接触内部结构
a25　加载哪些库，由宿主决定
a26　require 加载过的模块
a27　会缓存在 package.loaded 里
a28　再次 require 时直接返回缓存
a29　lua.c 的 main，也只是这套 API 的一个使用者
a30　它建立状态，在保护模式下调用 pmain
a31　pmain 打开标准库，把回收器切换到分代模式
a32　然后加载并运行脚本
a33　最后由 lua_close 释放所有对象
