==== 按名字找 Alice：没有索引，只能用 SCAN 把 4096 个键都看一遍（服务端的 Lua 脚本） ====
127.0.0.1:7379> CONFIG SET slowlog-log-slower-than 0
OK
local cur='0' local n=0 local found=false
repeat
  local r = redis.call('SCAN', cur, 'COUNT', 100)
  cur = r[1]
  for _, k in ipairs(r[2]) do
    n = n + 1
    if redis.call('HGET', k, 'name') == 'Alice' then found = k end
  end
until cur == '0'
return {found, n}
