
==== 建表 ====

:) CREATE DATABASE IF NOT EXISTS demo


:) DROP TABLE IF EXISTS demo.users8


:) DROP TABLE IF EXISTS demo.users8w


:) CREATE TABLE demo.users8w (id UInt32, name String, age UInt8) ENGINE = MergeTree ORDER BY id SETTINGS min_bytes_for_wide_part = 0


:) CREATE TABLE demo.users8 (id UInt32, name String, age UInt8) ENGINE = MergeTree ORDER BY id
