
==== 建表 ====

:) DROP TABLE IF EXISTS demo.users SYNC


:) CREATE TABLE demo.users (id UInt32, name String, age UInt8) ENGINE = MergeTree ORDER BY id


:) SHOW CREATE TABLE demo.users

CREATE TABLE demo.users
(
    `id` UInt32,
    `name` String,
    `age` UInt8
)
ENGINE = MergeTree
ORDER BY id
SETTINGS index_granularity = 8192
