==== 写入 4096 个用户（不写 _id，由驱动生成） ====

lab> 
const docs = [];
for (let i = 1; i <= 4096; i++) {
  docs.push(i === 2050
    ? { name: "Alice", age: 42 }
    : { name: "User" + String(i).padStart(5, "0") + "-abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMN", age: 18 + i % 63 });
}
db.users.insertMany(docs).insertedIds.length

lab> db.users.createIndex({ name: 1 })
name_1

lab> db.users.countDocuments()
4096


==== Alice 的文档、字段类型与 BSON 大小 ====

lab> db.users.findOne({ name: "Alice" })
{ _id: ObjectId('6afa9e875b27ce1a481bcdd3'), name: 'Alice', age: 42 }

lab> db.users.aggregate([{ $match: { name: "Alice" } }, { $project: { _id: { $type: "$_id" }, name: { $type: "$name" }, age: { $type: "$age" } } }]).toArray()
[ { _id: 'objectId', name: 'string', age: 'int' } ]

lab> bsonsize(db.users.findOne({ name: "Alice" }))

lab> bsonsize(db.users.findOne({ name: "User02049-abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMN" }))
